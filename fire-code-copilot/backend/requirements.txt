# --- Web API ---
fastapi
uvicorn[standard]
pydantic
pydantic-settings
python-dotenv

# --- LLM + embeddings ---
# To run an open-source model DIRECTLY (a .gguf via llama.cpp, or MLX on Apple Silicon),

anthropic              # Claude API (optional escalation only)

# --- PDF parsing ---
pymupdf                # primary extractor (import as: fitz) — keeps layout/structure
pypdf                  # fallback extractor

# --- Utilities ---
pyyaml                 # read config/code_cycles.yaml
tiktoken               # token-aware chunking
python-dateutil        # code-cycle date math

# --- Dev / tests ---
pytest                 # offline test suite (chunking, citations, agent, retrieval, cycles)
httpx                  # FastAPI TestClient (API smoke tests)
hypothesis             # property tests for chunking invariants
